        p = rightChild(p);

void Mid(Tree t) {
    if(t == NULL)
        return;
    Stack s = Create();
    Tree p = t;
    do{
        while(p != NULL) {
            Push(s, p);
            p = leftChild(p);
        }
        p = top(s);
        visit(p);
        p = rightChild(p);
        Pop(s);
    } while(!Empty(s) || p != NULL);
}

void Post(Tree t) {
    if(t == NULL) 
        return ;
    Stack s = Create();
    Tree pre = NULL, p = t;
    do{
        while(p != NULL) {
            Push(s, p);
            p = leftChild(p);
        }
        p = top(s);
        if(rightChild(p) != NULL && rightChild(p) != pre) {
            p = rightChild(p);
        }
        else {
            visit(p);
            pre = p;
            p = NULL;
            Pop(s);
        }

    if(p >= 0 && p < t->n) {
        for(int i = p + 1; i < t->n; i++) {
            if(t->list[i].parent == t->list[p].parent)
                return i;
        }

}

int leftChild(tree t, int p) {
    if(t->list[p + 1].parent == p)
        return p + 1;
    else
        return -1;
}
